## 1. 项目概述

### 1.1 项目简介

mfish-nocode-view 是一个基于 Vue 3 + TypeScript 的低代码可视化大屏开发平台，采用 monorepo 架构管理多个内部包和业务包。项目集成了业界主流的前端技术栈，包括 Vite 7 构建工具、Pinia 状态管理、Vue Router 4 路由管理、Ant Design Vue 4 组件库以及 ECharts 6 数据可视化库，为企业级应用提供完整的前端解决方案。

### 1.2 技术栈概览

```
核心框架: Vue 3.5.29 + TypeScript 5.9.3
构建工具: Vite 7.3.1 + Turbo 2.8.10
状态管理: Pinia 3.0.4
路由管理: Vue Router 4.6.4
UI组件库: Ant Design Vue 4.2.6 + Naive UI 2.43.2
数据可视化: ECharts 6.0.0 + echarts-gl 2.0.9
HTTP请求: Axios 1.13.5
国际化: vue-i18n 11.2.8
样式预处理器: Less 4.5.1
包管理器: pnpm 10.28.1
代码规范: ESLint 9.39.3 + Prettier 3.8.1 + Stylelint 16.26.1
```

  async login(params: LoginParams) {
    const { access_token, refresh_token } = await loginApi(params);
    this.setToken(access_token);
    this.setRefreshToken(refresh_token);
    return await this.getAccountInfo();
  },
  
  // 获取用户信息并构建路由
  async getAccountInfo(): Promise<SsoUser> {
    const userInfo = await this.getUserInfoAction();
    const permissionStore = usePermissionStore();
    permissionStore.setPermissions(new Set(userInfo.permissions));
    if (!permissionStore.isDynamicAddedRoute) {
      await permissionStore.addRouter(router);
    }
    return userInfo;
  },
  
  // 登出
  async logout() {
    await doLogout();
    this.setToken(undefined);
    clearAuthCache(true);
    await router.replace(PageEnum.BASE_LOGIN);
  }

// 核心算法：菜单转路由

const buildRoute = (menu: MenuListItem): AppRouteRecordRaw => {
  const route: AppRouteRecordRaw = {
    name: menu.id,
    path: formatPath(menu.routePath),
    meta: {
      title: menu.menuName,
      icon: menu.menuIcon,
      hideMenu: !menu.isVisible,
      keepAlive: !!menu.isKeepalive
    }
  };
  
  switch (menu.menuType) {
    case MenuType.目录:
      route.component = LAYOUT;  // 目录使用 Layout 包装
      break;
    case MenuType.菜单:
      if (isUrl(menu.component)) {
        route.meta.frameSrc = menu.component;
        if (!menu.isExternal) {
          route.component = IFRAME;  // 内部 iframe 打开
        }
      } else {
        route.component = importComponent(menu.component);
      }
      break;
  }
  return route;
};

## 8. API 接口设计

### 8.1 HTTP 请求封装

```typescript
// packages/core/src/utils/http/axios/index.ts
const transform: AxiosTransform = {
  // 请求前拦截
  beforeRequestHook: (config, options) => {
    // 1. 拼接 API 地址
    if (!config.url?.startsWith('http')) {
      config.url = `${apiUrl}${config.url}`;
    }
    // 2. GET 请求添加时间戳
    if (config.method === 'GET') {
      config.params = { ...config.params, _t: Date.now() };
    }
    // 3. 非GET请求处理参数
    if (config.method !== 'GET' && Object.keys(config.data || {}).length) {
      config.data = config.data;
      config.params = config.params;
    }
    return config;
  },
  
  // 请求拦截器（添加Token）
  requestInterceptors: (config, options) => {
    const token = getToken();
    if (token) {
      config.headers.Authorization = `Bearer ${token}`;
    }
    return config;
  },
  
  // 响应拦截器
  responseInterceptors: (res) => res,
  
  // 响应错误处理
  responseInterceptorsCatch: (axiosInstance, error) => {
    const errorLogStore = useErrorLogStoreWithOut();
    errorLogStore.addAjaxErrorInfo(error);
    
    // 检查是否需要重试
    const retryRequest = new AxiosRetry();
    if (config.method === 'GET' && isOpenRetry) {
      retryRequest.retry(axiosInstance, error);
    }
    return Promise.reject(error);
  }
};
```

    }
    return this.request({ ...config, method: 'GET' }, options);
  }

    const formData = new FormData();
    Object.keys(params).forEach(key => {
      formData.append(key, params[key]);
    });
    return this.request({
      ...config,
      timeout: 30 * 60 * 1000,
      method: 'POST',
      data: formData,
      headers: { 'Content-Type': 'multipart/form-data' }
    }, options);
  }
  
  // 文件下载（30分钟超时）
  download<T>(config, options?: RequestOptions): Promise<T> {
    return this.request({
      ...config,
      method: 'get',
      responseType: 'blob',
      timeout: 30 * 60 * 1000
    }, options, transform.downloadResponseHook);
  }
}

// packages/core/src/utils/helper/TreeHelper.ts

// 列表转树（O(n)复杂度，使用Map优化）
export function listToTree<T>(list: any[], config = {}): T[] {
  const { id, children, pid } = getConfig(config);
  const nodeMap = new Map();
  const result: T[] = [];
  
  // 第一遍：建立 id -> node 的映射
  for (const node of list) {
    node[children] = node[children] || [];
    nodeMap.set(node[id], node);
  }
  
  // 第二遍：构建树关系
  for (const node of list) {
    const parent = nodeMap.get(node[pid]);
    (parent ? parent[children] : result).push(node);
  }
  return result;
}

// 树转列表
export function treeToList<T>(tree: any, config = {}): T {
  const { children } = getConfig(config);
  const result = [...tree];
  for (let i = 0; i < result.length; i++) {
    if (!result[i][children]) continue;
    result.splice(i + 1, 0, ...result[i][children]);
  }
  return result;
}

// 查找节点路径
export function findPath<T>(tree, func, config = {}): T[] | null {
  const path: T[] = [];
  const list = [...tree];
  const visitedSet = new Set();
  
  while (list.length > 0) {
    const node = list[0];
    if (visitedSet.has(node)) {
      path.pop();
      list.shift();
    } else {
      visitedSet.add(node);
      node[children] && list.unshift(...node[children]);
      path.push(node);
      if (func(node)) return path;
    }
  }
  return null;
}

// 树形结构遍历（可中断）
export function forEach<T>(tree, func, config = {}): void {
  const list = [...tree];
  for (let i = 0; i < list.length; i++) {
    if (func(list[i])) return;  // 返回 true 终止遍历
    list[i][children] && list.splice(i + 1, 0, ...list[i][children]);
  }
}

export function formatPath(path: string) {
  if (path.startsWith('/') || isUrl(path)) return path;
  return `/${path}`;

// 递归拼接父路径
function joinParentPath(menus: Menu[], parentPath = '') {
  for (const menu of menus) {
    if (!(menu.path.startsWith('/') || isUrl(menu.path))) {
      menu.path = `${parentPath}/${menu.path}`;
    }
    if (menu?.children?.length) {
      joinParentPath(
        menu.children,
        menu.meta?.hidePathForChildren ? parentPath : menu.path
      );
    }
  }
}

// 动态参数替换
const menuParamRegex = /:([\s\S]+?)((?=\/)|$)/g;

export function configureDynamicParamsMenu(menu: Menu, params: RouteParams) {
  let realPath = paramPath || path;
  const matchArr = realPath.match(menuParamRegex);
  
  matchArr?.forEach(it => {
    const key = it.slice(1);
    if (params[key]) {
      realPath = realPath.replace(`:${key}`, params[key] as string);
    }
  });
  
  menu.path = realPath;
  menu.children?.forEach(item => configureDynamicParamsMenu(item, params));
}

## 13. 潜在技术风险点

### 13.1 高风险项

| 风险 | 描述 | 建议 |
|------|------|------|
| echarts-gl 版本 | 使用 `echarts-gl 2.0.9` 可能与最新 echarts 6.x 存在兼容性问题 | 升级前进行完整测试 |
| 循环依赖 | Store 之间可能存在循环引用 | 审查 usePermissionStore 与 useUserStore 的依赖关系 |
| 大数据量 | 树形组件和表格未实现虚拟滚动 | 考虑引入虚拟列表优化 |
| 内存泄漏 | 定时器、事件监听未完全清理 | 使用 onUnmounted 确保清理 |

### 13.2 中风险项

| 风险 | 描述 | 建议 |
|------|------|------|
| Token 刷新 | refreshToken 逻辑可能存在并发问题 | 增加锁机制 |
| 路由守卫顺序 | 多个守卫的执行顺序依赖较严格 | 增加注释说明 |
| 第三方库版本 | catalog 模式锁定版本不严格 | 明确关键库版本约束 |

### 13.3 低风险项

| 风险 | 描述 | 建议 |
|------|------|------|
| 国际化遗漏 | 部分字符串可能未使用 i18n | 加强代码审查 |
| 样式覆盖 | Ant Design 样式修补可能导致升级问题 | 使用 CSS Variables |
| 错误边界 | 全局错误处理可能捕获预期错误 | 细化错误分类 |

### 13.4 安全建议

1. **XSS 防护**：用户输入内容在渲染前需做转义处理
2. **CSRF**：请求头中已包含 Authorization Token，需确保后端验证
3. **敏感信息**：避免在 localStorage 中存储敏感信息
4. **文件上传**：当前实现需后端做进一步安全验证
